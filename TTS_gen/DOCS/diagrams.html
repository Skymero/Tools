flowchart TD
    A["Start: parse CLI args\naudio, images_dir, timeline, output"] --> B{"Inputs exist?\naudio / images dir / timeline"}
    B -- no --> E1["FileNotFoundError\nprint error, return False, exit 1"]
    B -- yes --> C["Load timeline JSON\nrequire 'scenes' array, non-empty"]
    C -- invalid --> E2["ValueError\nreturn False, exit 1"]
    C -- valid --> D["mkdir temp_segments/"]
    D --> F["For each scene i:\nread image, start, end, motion, intensity"]
    F --> G{"image field present\nand file exists?"}
    G -- no --> E3["ValueError / FileNotFoundError\nabort run, exit 1"]
    G -- yes --> H["Loop image @30fps ->\nscale+pad to 1080x1920 portrait"]
    H --> I{"motion type?"}
    I -- push_in --> J1["zoompan zoom+intensity\nclamped 1.5"]
    I -- pull_out --> J2["zoompan zoom-intensity\nclamped 1.0"]
    I -- pan_left --> J3["zoompan x: iw-iw/zoom"]
    I -- pan_right --> J4["zoompan x: 0"]
    I -- static/other --> J5["fps filter only"]
    J1 & J2 & J3 & J4 & J5 --> K["Encode segment_i.mp4\nlibx264, yuv420p, crf23, t=duration"]
    K --> F
    F -- all scenes done --> L["Write concat.txt ->\nconcat demuxer -> video_only.mp4"]
    L --> M["ffprobe audio + video durations"]
    M --> N{"video shorter\nthan audio?"}
    N -- yes --> O["Loop video via concat\nceil(audio/video)+1 copies"]
    N -- no --> P["keep as-is (audio may\ntrim final output)"]
    O --> Q["Mux: video + AAC 192k audio\n-> output.mp4"]
    P --> Q
    Q --> R["Delete segments, concat.txt,\nvideo_only.mp4, temp dir"]
    R --> S["Return True, exit 0"]
